{{- /* Drawn when the site is built, as SVG, from the block's table: the first column names each point, the others are series of numbers. No script. The table stays for screen readers. */ -}} {{- $html := string .inner -}} {{- $rows := slice -}} {{- range findRE `(?s).*?` $html -}} {{- $r := slice -}} {{- range findRE `(?s)]*>.*?` . }}{{ $r = $r | append (. | plainify | htmlUnescape | strings.TrimSpace) }}{{ end -}} {{- $rows = $rows | append (dict "cells" $r) -}} {{- end -}} {{- if ge (len $rows) 2 -}} {{- $head := (index $rows 0).cells -}} {{- $data := after 1 $rows -}} {{- $names := after 1 $head -}} {{- $m := len $names -}} {{- if eq .s.kind "pie" }}{{ $m = 1 }}{{ end -}} {{- $n := len $data -}} {{- $num := dict -}} {{- $max := 0.0 -}} {{- $total := 0.0 -}} {{- range $i, $row := $data -}} {{- range $s := seq 0 (sub $m 1) -}} {{- $v := float (replaceRE `[^0-9.\-]` "" (index $row.cells (add $s 1) | default "0") | default "0") -}} {{- $num = merge $num (dict (printf "%d-%d" $s $i) $v) -}} {{- if gt $v $max }}{{ $max = $v }}{{ end -}} {{- if eq $s 0 }}{{ $total = add $total $v }}{{ end -}} {{- end -}} {{- end -}} {{- if le $max 0.0 }}{{ $max = 1.0 }}{{ end -}} {{- $k := math.Pow 10 (math.Floor (div (math.Log $max) (math.Log 10))) -}} {{- $top := mul (math.Ceil (div $max $k)) $k -}} {{- $W := 640.0 }}{{ $H := 320.0 }}{{ $L := 52.0 }}{{ $T := 16.0 }}{{ $R := 16.0 }}{{ $B := 44.0 -}} {{- $pw := sub $W (add $L $R) }}{{ $ph := sub $H (add $T $B) -}} {{- else -}} {{- end -}}
