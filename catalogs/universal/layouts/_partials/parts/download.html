{{- /* The block's first link is the file; the rest says what is in it. */ -}} {{- $html := string .inner -}} {{- $href := replaceRE `(?s)^.*?]*>(.*?).*$` "$1" $html -}} {{- $rest := replaceRE `(?s)^\s*

.*?

` "" $html -}} {{- $ext := path.Ext $href | strings.TrimPrefix "." | upper -}} {{- $size := "" -}} {{- $local := path.Join "static" $href -}} {{- if and (hasPrefix $href "/") (fileExists $local) -}} {{- $bytes := (os.Stat $local).Size -}} {{- $size = cond (ge $bytes 1048576) (printf "%.1f MB" (div (float $bytes) 1048576)) (printf "%.0f KB" (math.Ceil (div (float $bytes) 1024))) -}} {{- end -}}
{{ $name | safeHTML }} {{ delimit (slice $ext $size | complement (slice "")) " · " }} {{- with trim $rest " \n" }}
{{ . | safeHTML }}
{{- end }}
